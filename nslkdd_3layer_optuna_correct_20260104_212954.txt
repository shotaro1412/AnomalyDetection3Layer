# ============================================================
# 層割り当て関数
# ============================================================
def assign_layer(scores, threshold_b, threshold_c):
    layers = np.ones(len(scores), dtype=int)  # B層
    layers[scores >= threshold_b] = 0  # A層
    layers[scores <= threshold_c] = 2  # C層
    return layers

# ============================================================
# Optuna目的関数（Train内で交差検証）
# ============================================================
def objective(trial):
    q_b = trial.suggest_int('q_b', 10, 50)
    q_c = trial.suggest_int('q_c', 1, 15)

    skf = StratifiedKFold(n_splits=N_SPLITS, shuffle=True, random_state=RANDOM_STATE)
    f1_scores = []
    prec_scores = []
    rec_scores = []
    auc_scores = []

    for fold_idx, (train_idx, val_idx) in enumerate(skf.split(X_train, y_train)):
        X_tr, X_val = X_train[train_idx], X_train[val_idx]
        y_tr, y_val = y_train[train_idx], y_train[val_idx]

        # Isolation Forest
        X_tr_normal = X_tr[y_tr == 0]
        if len(X_tr_normal) > 30000:
            sample_idx = np.random.choice(len(X_tr_normal), 30000, replace=False)
            X_tr_normal_sample = X_tr_normal[sample_idx]
        else:
            X_tr_normal_sample = X_tr_normal

        iso_forest = IsolationForest(contamination=0.1, random_state=RANDOM_STATE, n_jobs=-1)
        iso_forest.fit(X_tr_normal_sample)

        tr_scores = iso_forest.score_samples(X_tr)
        val_scores = iso_forest.score_samples(X_val)
        normal_scores = iso_forest.score_samples(X_tr_normal_sample)

        threshold_b = np.percentile(normal_scores, 100 - q_b)
        threshold_c = np.percentile(normal_scores, q_c)

        tr_layers = assign_layer(tr_scores, threshold_b, threshold_c)
        val_layers = assign_layer(val_scores, threshold_b, threshold_c)

        X_tr_3layer = np.column_stack([X_tr, tr_scores, tr_layers])
        X_val_3layer = np.column_stack([X_val, val_scores, val_layers])

        model = xgb.XGBClassifier(
            scale_pos_weight=scale_pos_weight,
            random_state=RANDOM_STATE,
            eval_metric='logloss',
            n_jobs=-1,
            verbosity=0,
            max_depth=6,
            n_estimators=100
        )
        model.fit(X_tr_3layer, y_tr)
        y_pred = model.predict(X_val_3layer)
        y_pred_proba = model.predict_proba(X_val_3layer)[:, 1]

        # 各指標を計算
        fold_f1 = f1_score(y_val, y_pred)
        fold_prec = precision_score(y_val, y_pred)
        fold_rec = recall_score(y_val, y_pred)
        fold_auc = roc_auc_score(y_val, y_pred_proba)
        tn, fp, fn, tp = confusion_matrix(y_val, y_pred).ravel()

        f1_scores.append(fold_f1)
        prec_scores.append(fold_prec)
        rec_scores.append(fold_rec)
        auc_scores.append(fold_auc)

        # 各foldの指標を保存
        trial.set_user_attr(f'fold{fold_idx+1}_f1', fold_f1)
        trial.set_user_attr(f'fold{fold_idx+1}_precision', fold_prec)
        trial.set_user_attr(f'fold{fold_idx+1}_recall', fold_rec)
        trial.set_user_attr(f'fold{fold_idx+1}_auc', fold_auc)
        trial.set_user_attr(f'fold{fold_idx+1}_tp', int(tp))
        trial.set_user_attr(f'fold{fold_idx+1}_fp', int(fp))
        trial.set_user_attr(f'fold{fold_idx+1}_fn', int(fn))
        trial.set_user_attr(f'fold{fold_idx+1}_tn', int(tn))

    trial.set_user_attr('f1_std', np.std(f1_scores))
    trial.set_user_attr('precision_mean', np.mean(prec_scores))
    trial.set_user_attr('recall_mean', np.mean(rec_scores))
    trial.set_user_attr('auc_mean', np.mean(auc_scores))
    return np.mean(f1_scores)

# ============================================================
# 交差検証評価関数（XGBoost単体と3層構造の両方を評価）
# ============================================================
def evaluate_with_cv(q_b, q_c):
    """最適な分位点を使って交差検証で評価"""
    skf = StratifiedKFold(n_splits=N_SPLITS, shuffle=True, random_state=RANDOM_STATE)

    # XGBoost単体の結果格納
    base_results = {
        'f1': [], 'precision': [], 'recall': [], 'auc': [],
        'tp': [], 'fp': [], 'fn': [], 'tn': []
    }

    # 3層構造の結果格納
    layer3_results = {
        'f1': [], 'precision': [], 'recall': [], 'auc': [],
        'tp': [], 'fp': [], 'fn': [], 'tn': []
    }

    for fold_idx, (train_idx, val_idx) in enumerate(skf.split(X_train, y_train)):
        X_tr, X_val = X_train[train_idx], X_train[val_idx]
        y_tr, y_val = y_train[train_idx], y_train[val_idx]

        # ========== XGBoost単体 ==========
        model_base = xgb.XGBClassifier(
            scale_pos_weight=scale_pos_weight,
            random_state=RANDOM_STATE,
            eval_metric='logloss',
            n_jobs=-1,
            verbosity=0,
            max_depth=6,
            n_estimators=100
        )
        model_base.fit(X_tr, y_tr)
        y_pred_base = model_base.predict(X_val)
        y_pred_proba_base = model_base.predict_proba(X_val)[:, 1]

        tn, fp, fn, tp = confusion_matrix(y_val, y_pred_base).ravel()
        base_results['f1'].append(f1_score(y_val, y_pred_base))
        base_results['precision'].append(precision_score(y_val, y_pred_base))
        base_results['recall'].append(recall_score(y_val, y_pred_base))
        base_results['auc'].append(roc_auc_score(y_val, y_pred_proba_base))
        base_results['tp'].append(int(tp))
        base_results['fp'].append(int(fp))
        base_results['fn'].append(int(fn))
        base_results['tn'].append(int(tn))

        # ========== 3層構造 ==========
        X_tr_normal = X_tr[y_tr == 0]
        if len(X_tr_normal) > 30000:
            sample_idx = np.random.choice(len(X_tr_normal), 30000, replace=False)
            X_tr_normal_sample = X_tr_normal[sample_idx]
        else:
            X_tr_normal_sample = X_tr_normal

        iso_forest = IsolationForest(contamination=0.1, random_state=RANDOM_STATE, n_jobs=-1)
        iso_forest.fit(X_tr_normal_sample)

        tr_scores = iso_forest.score_samples(X_tr)
        val_scores = iso_forest.score_samples(X_val)
        normal_scores = iso_forest.score_samples(X_tr_normal_sample)

        threshold_b = np.percentile(normal_scores, 100 - q_b)
        threshold_c = np.percentile(normal_scores, q_c)

        tr_layers = assign_layer(tr_scores, threshold_b, threshold_c)
        val_layers = assign_layer(val_scores, threshold_b, threshold_c)

        X_tr_3layer = np.column_stack([X_tr, tr_scores, tr_layers])
        X_val_3layer = np.column_stack([X_val, val_scores, val_layers])

        model_3layer = xgb.XGBClassifier(
            scale_pos_weight=scale_pos_weight,
            random_state=RANDOM_STATE,
            eval_metric='logloss',
            n_jobs=-1,
            verbosity=0,
            max_depth=6,
            n_estimators=100
        )
        model_3layer.fit(X_tr_3layer, y_tr)
        y_pred_3layer = model_3layer.predict(X_val_3layer)
        y_pred_proba_3layer = model_3layer.predict_proba(X_val_3layer)[:, 1]

        tn, fp, fn, tp = confusion_matrix(y_val, y_pred_3layer).ravel()
        layer3_results['f1'].append(f1_score(y_val, y_pred_3layer))
        layer3_results['precision'].append(precision_score(y_val, y_pred_3layer))
        layer3_results['recall'].append(recall_score(y_val, y_pred_3layer))
        layer3_results['auc'].append(roc_auc_score(y_val, y_pred_proba_3layer))
        layer3_results['tp'].append(int(tp))
        layer3_results['fp'].append(int(fp))
        layer3_results['fn'].append(int(fn))
        layer3_results['tn'].append(int(tn))

    return base_results, layer3_results
